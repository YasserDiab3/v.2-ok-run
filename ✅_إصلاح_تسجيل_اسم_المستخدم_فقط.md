### 1️⃣ في `Backend/Clinic.gs`

#### أ. إضافة دالة مساعدة `getUserNameFromDatabase_`:

```javascript
/**
 * الحصول على اسم المستخدم من قاعدة البيانات بناءً على email أو id
 * @param {string} email - البريد الإلكتروني للمستخدم
 * @param {string} userId - معرف المستخدم
 * @return {string} اسم المستخدم أو null
 */
function getUserNameFromDatabase_(email, userId) {
    try {
        if (!email && !userId) {
            return null;
        }
        
        const users = readFromSheet('Users', getSpreadsheetId());
        if (!users || !Array.isArray(users) || users.length === 0) {
            Logger.log('⚠️ [BACKEND] لا توجد بيانات مستخدمين في قاعدة البيانات');
            return null;
        }
        
        // البحث عن المستخدم بناءً على email أو id
        const user = users.find(u => {
            const userEmail = (u.email || '').toString().toLowerCase().trim();
            const searchEmail = email ? email.toLowerCase().trim() : '';
            const userIdFromDb = (u.id || '').toString().trim();
            const searchUserId = userId ? userId.toString().trim() : '';
            
            return (email && userEmail === searchEmail) || (userId && userIdFromDb === searchUserId);
        });
        
        if (user) {
            const userName = (user.name || user.displayName || '').toString().trim();
            if (userName && userName !== 'النظام' && userName !== '') {
                Logger.log('✅ [BACKEND] تم العثور على اسم المستخدم من قاعدة البيانات: ' + userName);
                return userName;
            }
        }
        
        Logger.log('⚠️ [BACKEND] لم يتم العثور على اسم المستخدم في قاعدة البيانات');
        return null;
    } catch (error) {
        Logger.log('❌ [BACKEND] خطأ في getUserNameFromDatabase_: ' + error.toString());
        return null;
    }
}
```

**ما يحدث:**
- ✅ البحث عن المستخدم في قاعدة البيانات بناءً على `email` أو `id`
- ✅ استخراج `name` أو `displayName` من بيانات المستخدم
- ✅ إرجاع اسم المستخدم فقط (وليس email)
